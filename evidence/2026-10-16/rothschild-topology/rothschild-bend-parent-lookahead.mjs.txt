import{readFile,writeFile}from'node:fs/promises';

rows.push({time:row.time,speed:car.speed,progress:car.progress,lateral,nominalLookMetres:nominal,actualCenterlineArcMetres:actualArc,chordMetres:Math.hypot(tx-car.x,tz-car.z),err,input:row.input});}
const result={scope:'Read-only analysis of recorded actual parent states against unchanged Rothschild geometry; not current-head race acceptance',captureHead:data.sourceHead,trackLength:track.length,rows};await writeFile(new URL('./rothschild-bend-parent-lookahead.json',import.meta.url),JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result,null,2));
